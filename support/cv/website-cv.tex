\cvsection{Research Systems \& Open-Source Platforms}
\cvsummary{Runtime, shared-infrastructure, and sim-to-real systems across embedded computers
and two full-scale vehicles.}
\begin{itemize}
    \item \textbf{\href{https://autoware.org/about/why-autoware/}{Autoware Foundation leadership}.} API Workgroup Lead; Technical Steering Committee and Strategic Planning Committee member. Defines APIs, middleware, V2X interoperability, and development priorities for the world's largest open-source autonomous-driving software stack, used by 500+ companies and deployed across 30 vehicle types in 20 countries.
    \item \textbf{SIM/DAVOS/DaVoS.} Designed newest-frame shared-memory sensor transport for ROS~2/Autoware. Reduced maximum observed latency by up to 98\% and p99 by approximately 96\%; the measured time saving equals 13.6~ft of travel at 40~mph, before accounting for downstream processing or braking. \textbf{Outputs:} \href{https://arxiv.org/abs/2510.11448}{SIM}; \href{https://arxiv.org/abs/2601.05072}{DAVOS}; \href{https://doi.org/10.1109/MOST69733.2026.00043}{DaVoS}; \href{https://github.com/Croquembouche/DAVOS_SIM}{code}.
    \item \textbf{IPI/InfraSense.} Designed a CV/CAV API and authenticated roadway gateway; evaluated C-V2X/\allowbreak private~5G under mobility, load, weak signal, and restarts. \textbf{Outputs:} Edge4AV; \href{https://github.com/Croquembouche/IPI}{IPI}; InfraSense.
    \item \textbf{BlueICE/CARScenes/S-Score.} Built tools for sim-to-real validation, scenario data, and dataset coverage. \textbf{Outputs:} \href{https://arxiv.org/abs/2405.01328}{BlueICE}; \href{https://arxiv.org/abs/2511.10701}{CARScenes}; \href{https://weisongshi.org/papers/he26-udrive.pdf}{S-Score}.
\end{itemize}
